function CheckItemEquipped {
    .@itemIdCurrent = getequipid(getarg(0));
    .@refineLevelCurrent = getequiprefinerycnt(getarg(1));
    if(getarg(1) != .@itemIdCurrent && getarg(2) != .@refineLevelCurrent){
        next;
        mes "Hey! Don't touch it while I'm still working on it!";
        return 0;
    }
    return 1;
}
